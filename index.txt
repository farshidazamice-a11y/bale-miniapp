<div id="debug-info" style="position:fixed; bottom:0; left:0; right:0; background:#fffbe6; padding:10px; font-size:12px; border-top:1px solid #ccc; z-index:9999; direction:ltr; text-align:left;">
  <strong>Debug Info:</strong><br>
  window.Bale: <span id="dbg-bale"></span><br>
  window.Bale.WebApp: <span id="dbg-webapp"></span><br>
  initData: <span id="dbg-initdata"></span><br>
  initDataUnsafe: <span id="dbg-unsafe"></span>
</div>

<script>
  // این کد را در انتهای تگ body، بعد از اسکریپت اصلی خود قرار دهید
  document.addEventListener('DOMContentLoaded', function() {
    const webApp = window.Bale?.WebApp || window.Telegram?.WebApp;
    
    document.getElementById('dbg-bale').textContent = typeof window.Bale;
    document.getElementById('dbg-webapp').textContent = webApp ? '✅ وجود دارد' : '❌ وجود ندارد';
    
    if (webApp) {
      document.getElementById('dbg-initdata').textContent = webApp.initData ? `✅ طول: ${webApp.initData.length}` : '❌ خالی';
      document.getElementById('dbg-unsafe').textContent = webApp.initDataUnsafe ? '✅ دارد' : '❌ ندارد';
    } else {
      document.getElementById('dbg-initdata').textContent = 'N/A';
      document.getElementById('dbg-unsafe').textContent = 'N/A';
    }
  });
</script>
