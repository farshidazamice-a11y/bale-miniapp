  <script>
    (function() {
      // بررسی ۱: آیا SDK لود شده؟
      const sdkStatus = document.getElementById('sdk-status');
      const baleType = typeof window.Bale;
      const tgType = typeof window.Telegram;
      
      if (baleType === 'undefined' && tgType === 'undefined') {
        sdkStatus.innerHTML = '<span class="no">❌ SDK بارگذاری نشد</span><br>' +
          '<code>window.Bale: ' + baleType + '</code><br>' +
          '<code>window.Telegram: ' + tgType + '</code>';
      } else {
        sdkStatus.innerHTML = '<span class="ok">✅ SDK بارگذاری شد</span><br>' +
          '<code>window.Bale: ' + baleType + '</code><br>' +
          '<code>window.Telegram: ' + tgType + '</code>';
      }
      
      // بررسی ۲: آبجکت WebApp
      const webApp = window.Bale?.WebApp || window.Telegram?.WebApp;
      const webappStatus = document.getElementById('webapp-status');
      
      if (!webApp) {
        webappStatus.innerHTML = '<span class="no">❌ WebApp وجود ندارد</span>';
        document.getElementById('initdata-status').innerHTML = '<span class="no">—</span>';
        return;
      }
      
      webappStatus.innerHTML = '<span class="ok">✅ WebApp وجود دارد</span><br>' +
        '<code>platform: ' + (webApp.platform || 'نامشخص') + '</code><br>' +
        '<code>version: ' + (webApp.version || 'نامشخص') + '</code>';
      
      // بررسی ۳: initData
      webApp.ready?.();
      webApp.expand?.();
      
      const initData = webApp.initData;
      const initDataStatus = document.getElementById('initdata-status');
      
      if (!initData) {
        initDataStatus.innerHTML = '<span class="no">❌ initData خالی است</span><br>' +
          '<small>یعنی بله داده‌ای تزریق نکرده</small>';
      } else {
        initDataStatus.innerHTML = '<span class="ok">✅ initData وجود دارد</span><br>' +
          '<code>طول: ' + initData.length + ' کاراکتر</code><br>' +
          '<code>' + initData.substring(0, 80) + '...</code>';
      }
    })();
  </script>
