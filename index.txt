  <script>
    const APPS_SCRIPT_URL = 'https://script.google.com/macros/s/AKfycbyrIqUqRReosGJch2dOYfv2g7LTpiEDUwdlkCFGrpqiGwvC_70NWnFCgRlQYbxNgy48/exec';
    
    (async function init() {
      const loadingEl = document.getElementById('loading');
      const errorEl = document.getElementById('error');
      const mainEl = document.getElementById('main');
      const errorMsgEl = document.getElementById('error-message');
      const debugInfoEl = document.getElementById('debug-info');
      
      try {
        const webApp = window.Bale?.WebApp || window.Telegram?.WebApp;
        
        // --- دیباگ: وضعیت SDK ---
        const debugData = {
          'window.Bale': typeof window.Bale,
          'window.Bale?.WebApp': typeof window.Bale?.WebApp,
          'initData': webApp?.initData ? `"${webApp.initData.substring(0, 50)}..."` : 'خالی یا undefined',
          'initDataUnsafe': typeof webApp?.initDataUnsafe,
        };
        console.log('Debug Info:', debugData);

        if (!webApp) {
          throw new Error('SDK مینی‌اپ بارگذاری نشد.');
        }
        
        webApp.ready?.();
        webApp.expand?.();
        
        const initData = webApp.initData;
        if (!initData) {
          // نمایش اطلاعات دیباگ روی صفحه
          errorMsgEl.textContent = 'initData موجود نیست.';
          debugInfoEl.classList.remove('hidden');
          debugInfoEl.innerHTML = '<strong>اطلاعات دیباگ:</strong><br>' + JSON.stringify(debugData, null, 2);
          throw new Error('initData is empty');
        }
        
        const response = await fetch(APPS_SCRIPT_URL + '?action=auth', {
          method: 'POST',
          headers: { 'Content-Type': 'text/plain;charset=utf-8' },
          body: JSON.stringify({ initData }),
        });
        
        const result = await response.json();
        if (!result.ok) throw new Error(result.error || 'احراز هویت ناموفق بود');
        renderUser(result.user);
        
      } catch (err) {
        loadingEl.classList.add('hidden');
        errorEl.classList.remove('hidden');
        if (!errorMsgEl.textContent) errorMsgEl.textContent = err.message;
        console.error('MiniApp init error:', err);
      }
    })();
    
    function renderUser(user) {
      document.getElementById('loading').classList.add('hidden');
      document.getElementById('main').classList.remove('hidden');
      document.getElementById('welcome-name').textContent = `👋 ${user.fullName}`;
      document.getElementById('welcome-sub').textContent = user.position || '';
      const badge = document.getElementById('access-badge');
      badge.textContent = user.accessLevel || 'نامشخص';
      document.getElementById('info-fullname').textContent = user.fullName || '—';
      document.getElementById('info-systemname').textContent = user.systemName || '—';
      document.getElementById('info-position').textContent = user.position || '—';
      document.getElementById('info-unit').textContent = user.unit || '—';
      document.getElementById('info-access').textContent = user.accessLevel || '—';
      document.getElementById('info-date').textContent = user.registerDate || '—';
    }
